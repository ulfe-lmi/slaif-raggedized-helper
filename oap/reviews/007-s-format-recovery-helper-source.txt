def read_json(path):

@dataclass(frozen=True)
class FrozenOrder:
    """Only the source argument: normal validator always sees approved bytes.
    Canonical target/other reads remain ordinary Paths, without monkeypatching IO.
    """
    path: Path
    data: bytes

    @property
    def name(self):
        return self.path.name

    @property
    def suffix(self):
        return self.path.suffix

    def is_file(self):
        return self.path.is_file()

    def is_symlink(self):
        return self.path.is_symlink()

    def read_bytes(self):
        return self.data


def exact_ci():
    raw = o.read_bytes(CI_SNAPSHOT)
    o.require(o.digest(raw) == CI_SNAPSHOT_HASH, 'Frozen CI reference changed')
    reference = json.loads(raw)
    expected = {j['name']: j for j in reference['jobs']}
    o.require(set(expected) == REQUIRED_JOBS and len(reference['jobs']) == 5,
              'Frozen required-job reference invalid')
    ci = o.gh('api', f'repos/{REPOSITORY}/actions/runs/{RUN}')
    page = o.gh('api', f'repos/{REPOSITORY}/actions/runs/{RUN}/jobs?per_page=100&filter=all')
    o.require(ci['id'] == RUN and ci['workflow_id'] == 403803560
              and ci['path'] == '.github/workflows/oap.yml' and ci['run_attempt'] == 1
              and ci['event'] == 'pull_request' and ci['head_branch'] == BRANCH
              and ci['head_sha'] == HEAD and ci['status'] == 'completed'
              and ci['conclusion'] == 'success', 'Exact current workflow/run identity changed')
    jobs = page['jobs']
    o.require(page['total_count'] == len(jobs) == 5
              and len({j['name'] for j in jobs}) == 5
              and {j['name'] for j in jobs} == REQUIRED_JOBS,
              'Missing/duplicate/unexpected required CI job or incomplete page')
    for job in jobs:
        wanted = expected[job['name']]
        o.require(job['id'] == wanted['databaseId'] and job['run_id'] == RUN
                  and job['status'] == 'completed' and job['conclusion'] == 'success',
                  'Required job identity/status changed')
        o.require([(s['number'], s['name']) for s in job['steps']]
                  == [(s['number'], s['name']) for s in wanted['steps']]
                  and all(s['status'] == 'completed' and s['conclusion'] == 'success'
                          for s in job['steps']), 'Required step missing/skipped/changed/not successful')
    o.require(sum(len(j['steps']) for j in jobs) == 78, 'Required-step cardinality changed')


def exact_rejected_graph(repo, repository, identifier):
    """Graph reconciliation ONLY: strict original report verifier still refuses."""
    o.require(repo == R and repository == REPOSITORY and identifier == '007-r',
              'Only exact rejected predecessor is recoverable')
    o.require(o.unique(R, 'orders', identifier) == R / ORDER
              and o.unique(R, 'reports', identifier) == R / REPORT,
              'Ambiguous or missing original order/report')
    data = o.read_bytes(R / REPORT)
    o.require(len(data) == 7426 and o.digest(data) == REPORT_HASH, 'Frozen rejected report changed')
    m = o.metadata(data.decode('utf-8'))
    o.require(m['Identifier'] == identifier and m['Repository'] == REPOSITORY
              and m['PR number'] == '9' and m['Branch'] == BRANCH and m['Base branch'] == 'main'
              and m['Result'] == 'PARTIAL' and m['Report publication commit'] == 'SELF'
              and m['Round base commit'] == PRIOR and m['PR main base'] == MAIN
              and m['Implementation head SHA'] == IMPL + ' (sole parent ad835959)',
              'Only exact recorded report/format defect is recoverable')
    o.commit_id(R, IMPL)
    try:
        STRICT_VERIFY(R, repository, identifier)
    except o.ProtocolError as error:
        o.require(str(error) == 'Full literal Git object ID required', 'Unexpected strict failure')
    else:
        raise o.ProtocolError('Expected strict failure disappeared; use normal protocol')
    pr = o.gh('pr', 'view', '9', '--repo', repository, '--json',
              'number,state,isDraft,headRefName,baseRefName,headRefOid,autoMergeRequest')
    o.require(pr['number'] == 9 and pr['state'] == 'OPEN' and not pr['isDraft']
              and pr['headRefName'] == BRANCH and pr['baseRefName'] == 'main'
              and pr['headRefOid'] == HEAD and pr['autoMergeRequest'] is None,
              'Current exact PR/branch/head changed')
    candidates = [p for p in o.remote_prs(repository) if p['headRefName'].startswith('oap/007-')]
    o.require(len(candidates) == 1 and candidates[0]['number'] == 9, 'Objective PR not unique')
    o.require(o.git(R, 'rev-parse', 'HEAD') == HEAD and not o.git(R, 'status', '--porcelain'),
              'Coding head/dirty state changed')
    o.require(o.git(R, 'show', '-s', '--format=%P', HEAD) == IMPL
              and o.git(R, 'diff-tree', '--no-commit-id', '--name-status', '-r', HEAD)
              == 'A\t' + REPORT and not o.git(R, 'ls-tree', '--name-only', IMPL, '--', REPORT),
              'Report is not new-file sole-parent/report-only SELF')
    o.require(o.command(['git', '-C', str(R), 'show', HEAD + ':' + REPORT]) == data,
              'Git report bytes differ')
    for relative in ('oap/active', ORDER):
        o.require(o.command(['git', '-C', str(R), 'show', IMPL + ':' + relative])
                  == o.read_bytes(R / relative), 'Original activation bytes changed')
    o.require(o.digest(o.read_bytes(R / ORDER)) == ORDER_HASH and o.active(R) == identifier,
              'Frozen prior activation changed')
    o.require(o.gh('api', f'repos/{repository}/git/ref/heads/main')['object']['sha'] == MAIN,
              'Default branch changed')
    exact_ci()
    return {'identifier': identifier, 'report_commit': HEAD, 'report_sha256': REPORT_HASH,
            'status': 'REJECTED_FORMAT_GRAPH_ONLY_RECONCILED_NOT_ACCEPTANCE'}

              and 'xhigh' in footer and 'Pursuing goal' not in tail,

    links = []
    for fd in (proc / 'fd').iterdir():

    o.require(str(rollout) in links and str(Path('/home/janezp/.codex/thread-writer-locks')
              / (tid + '.lock')) in links, 'Live designated process is not bound to pinned thread')

def publish_exact_next(source):
    attempt = S / '.oap-state/format-recovery-007-r-to-007-s-attempt.json'
    result_path = S / '.oap-state/format-recovery-007-r-to-007-s-result.json'
    # All publication state is checked under the NORMAL helper publication lock.
    with o.lock(S / '.oap-state/response-reader.lock'), o.lock(S / '.oap-state/control-reader.lock'), \
            o.lock(R / '.oap-state/publication.lock'):
        o.require(not attempt.exists() and not result_path.exists(),
                  'Prior recovery attempted; explicitly reconcile partial state, never replay')
        approved = read_json(S / 'audit/007-r-format-recovery-approved.json')
        o.require(approved['decision'] == 'APPROVED_ONE_EXACT_NEXT_SUFFIX_PUBLICATION'
                  and approved['script_sha256'] == o.digest(o.read_bytes(Path(__file__)))
                  and approved['prior_report'] == HEAD, 'Exact independent Root approval absent')
        resolution = read_json(S / 'audit/007-r-review.json')
        timing = read_json(S / 'audit/007-r-resolution-timing-event.json')
        author = read_json(S / 'audit/007-r-terminal-author-root-reconciliation.json')
        retired = read_json(S / 'audit/007-r-owned-reader-retirement.json')
        o.require(resolution['status'] == 'REJECTED_REPORT_FORMAT_WHOLE007_PARTIAL'
                  and resolution['report'] == timing['report_commit'] == HEAD
                  and not resolution['development_acceptance'], 'Rejected-format resolution absent')
        raw_author = o.read_bytes(S / 'audit/007-r-coding-terminal-format-failure.json')
        o.require(author['source_sha256'] == o.digest(raw_author) and author['report_commit'] == HEAD
                  and author['strict_format_failed'] is True
                  and author['all_current_jobs_steps_success'] is True, 'Author terminal evidence mismatch')
        o.require(retired['identifier'] == '007-r' and retired['reader_pid'] == 3861339
                  and retired['terminated_owned_reader'] is True and retired['coding_pane_quiet'] is True
                  and retired['process_start_identity_verified'] is True,
                  'Original reader retirement evidence absent')
        config = o.load_runtime(S / 'runtime.json', R, operational=True)
        identity = current_quiet(config)
        frozen = FrozenOrder(source, o.read_bytes(source))
        o.require(o.digest(frozen.data) == approved['next_order_sha256'], 'Approved order bytes changed')
        m = o.validate_order(R, frozen)
        o.require(m['Identifier'] == '007-s' and m['Expected active'] == '007-r'
                  and m['Base commit'] == HEAD and m['Branch'] == BRANCH
                  and m['PR number'] == '9' and m['PR mode'] == 'AMEND_EXISTING_PR'
                  and o.active(R) == '007-r', 'Only exact next same007 suffix permitted')
        existing = o.unique(R, 'orders', '007-s', False)
        o.require(existing is None and o.unique(R, 'reports', '007-s', False) is None,
                  'Next suffix already exists; reconcile, do not replay')
        calls = []

        def rejected_transition_only(repo, repository, identifier):
            o.require(not calls, 'Repeated predecessor recovery')
            calls.append(exact_rejected_graph(repo, repository, identifier))
            return calls[-1]

        o.verify_report = rejected_transition_only
        try:
            facts = o.remote_transition(R, m, config['github_repository'])
        finally:
            o.verify_report = STRICT_VERIFY
        o.require(len(calls) == 1 and o.verify_report is STRICT_VERIFY, 'Strict function not restored')
        # Fresh local authority/quiescence immediately before durable intent/order/active writes.
        fresh = o.load_runtime(S / 'runtime.json', R, operational=True)
        o.require(fresh == config and o.active(R) == '007-r'
                  and o.git(R, 'rev-parse', 'HEAD') == HEAD and not o.git(R, 'status', '--porcelain'),
                  'Current authority/active/working state changed during reconciliation')
        current_quiet(fresh)
        # Reconcile remote identity and exact required checks again nearest the writes.
        near = o.gh('pr', 'view', '9', '--repo', REPOSITORY, '--json',
                    'state,isDraft,headRefName,baseRefName,headRefOid,autoMergeRequest')
        o.require(near['state'] == 'OPEN' and not near['isDraft']
                  and near['headRefName'] == BRANCH and near['baseRefName'] == 'main'
                  and near['headRefOid'] == HEAD and near['autoMergeRequest'] is None
                  and o.gh('api', f'repos/{REPOSITORY}/git/ref/heads/main')['object']['sha'] == MAIN,
                  'Remote changed nearest publication; abort and re-review')
        exact_ci()
        again = o.load_runtime(S / 'runtime.json', R, operational=True)
        o.require(again == fresh and o.active(R) == '007-r'
                  and o.git(R, 'rev-parse', 'HEAD') == HEAD and not o.git(R, 'status', '--porcelain'),
                  'Local authority/state changed during final remote read')
        current_quiet(again)
        journal = {'at': o.now(), 'state': 'ATTEMPTED_BEFORE_ORDER_WRITE', 'prior_report': HEAD,
                   'next_identifier': '007-s', 'order_sha256': o.digest(frozen.data),
                   'script_sha256': o.digest(o.read_bytes(Path(__file__))), 'identity': identity,
                   'no_FIFO_operations': True, 'prior_report_accepted': False}
        o.atomic_write(attempt, (json.dumps(journal, indent=2) + '\n').encode(), immutable=True, mode=0o600)
        target = o.safe_path(R, 'oap/orders/' + frozen.name)
        # Same atomic primitives and order-before-active sequence as normal publish_order.
        o.atomic_write(target, frozen.data, immutable=True)
        o.atomic_write(o.safe_path(R, 'oap/active'), b'007-s\n')
        o.require(o.read_bytes(target) == frozen.data and o.digest(o.read_bytes(target))
                  == approved['next_order_sha256'] and o.active(R) == '007-s', 'Publication read-back failed')
        result = {'status': 'PUBLISHED_EXACT_ONE_TIME_FAILED_FORMAT_CONTINUATION', 'at': o.now(),
                  'remote': facts, 'prior_rejected_report': calls[0], 'next_identifier': '007-s',
                  'order_sha256': o.digest(frozen.data), 'FIFO_operations': 0,
                  'persistent_verifier_changed': False, 'prior_report_accepted': False}
        o.atomic_write(result_path, (json.dumps(result, indent=2) + '\n').encode(), immutable=True, mode=0o600)
        return result
